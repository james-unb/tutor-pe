\item
\begin{itemize}
\item[a)] Sabemos que $$\sigma^2=\text{Var}(X_i) = \mathcal{E}\left(X_i^2\right)-[\mathcal{E}(X_i)]^2=\mathcal{E}\left(X_i^2\right)-\mu^2$$ portanto $\mathcal{E}\left(X_i^2\right)=\mu^2+\sigma^2.$

\vspace{0.3cm}

\noindent Analogamente, $$\frac{\sigma^2}{n}=\text{Var}\left(\overline{X}\right) = \mathcal{E}\left(\overline{X}^2\right)-\left[\mathcal{E}\left(\overline{X}\right)\right]^2=\mathcal{E}\left(\overline{X}^2\right)-\mu^2$$ portanto $\mathcal{E}\left(\overline{X}^2\right)=\mu^2+\dfrac{\sigma^2}{n}.$
\item[b)] 
\begin{eqnarray*}
\mathcal{E}(X_i\overline{X})&=& \frac{1}{n}\mathcal{E}(X_iX_1+X_2X_i+\cdots+X_i^2+\cdots+X_iX_n)\\
&=& \frac{1}{n}\left\{\mathcal{E}(X_iX_1)+\cdots+\mathcal{E}(X_i^2)+\cdots+\mathcal{E}(X_iX_n)\right\}\\
&=& \frac{1}{n}\left\{\mathcal{E}(X_i)\mathcal{E}(X_1)+\cdots+\mu^2+\sigma^2+\cdots+\mathcal{E}(X_i)\mathcal{E}(X_n)\right\}\\
&=& \mu^2+\frac{\sigma^2}{n}.
\end{eqnarray*}
\item[c)] Para saber se um estimador é viciado calculamos sua esperança:
\begin{eqnarray*}
\mathcal{E}(T)&=& \frac{1}{n}\sum_{i=1}^n\mathcal{E}\left(X_i-\overline{X}\right)^2\\
&=& \frac{1}{n}\sum_{i=1}^n\mathcal{E}\left(X_i^2-2X_i\overline{X}+\overline{X}^2\right)\\
&=& \frac{1}{n}\sum_{i=1}^n\left\{\mathcal{E}\left(X_i^2\right)-2\mathcal{E}\left(X_i\overline{X}\right)+\mathcal{E}\left(\overline{X}^2\right)\right\}\\
&=& \frac{1}{n}\sum_{i=1}^n\left\{\mu^2+\sigma^2-2\left(\mu^2+\frac{\sigma^2}{n}\right)+\mu^2+ \frac{\sigma^2}{n} \right\}\\
&=& \sigma^2-\frac{\sigma^2}{n}\\
&=& \sigma^2\left(\frac{n-1}{n}\right).
\end{eqnarray*}

\noindent Como $\mathcal{E}(T)\neq \sigma^2$, o estimador $T$ é viciado para a variância.
\item[d)] Para que um estimador seja não viciado para a variância, sua esperança deve ser $\sigma^2$. Observe que se passarmos o termo $(n-1)/n$ para o outro lado da igualdade obtida no {\bf item c)} multiplicando, teremos $$\frac{n}{n-1}\times \mathcal{E}\left(T\right)=\mathcal{E}\left(\frac{n}{n-1}\times T\right)=\sigma^2.$$ Então basta definirmos o estimador $$S^2 = \frac{n}{n-1}\times T = \sum_{i=1}^n\frac{\left(X_i-\overline{X}\right)^2}{n-1},$$ e $S^2$ é não viciado para a variância.

\vspace{0.3cm}

\noindent É importante notar que essa é razão para usarmos $S^2$ como variância amostral ao invés de usarmos $T$.
\end{itemize}

\item Como $X_i \sim \text{Exp}(\lambda)$, para $0<x_i<\infty$, $i=1,\ldots,n$, a verossimilhança do parâmetro $\lambda$ será dada por 
\begin{eqnarray*}
L(\lambda;x)&=& f_{X_1}(x_1)\cdots f_{X_n}(x_n)\\
&=& \lambda e^{-\lambda x_1}\cdots\lambda e^{-\lambda x_n}\\
&=& \lambda^n\exp\left\{-\lambda\sum_{i=1}^{n}x_i\right\}.
\end{eqnarray*}
Portanto a logverossimilhança é dada por
\begin{eqnarray*}
l(\lambda;x)&=& \log\lambda^n+\log\left[\exp\left\{-\lambda\sum_{i=1}^{n}x_i\right\}\right]\\
&=& n\log\lambda-\lambda\sum_{i=1}^{n}x_i.
\end{eqnarray*}
Derivando a logverossimilhança temos que $$l'(\lambda;x)=\frac{n}{\lambda}-\sum_{i=1}^{n}x_i=0 \iff \lambda=\frac{n}{\sum_{i=1}^{n}x_i}=\frac{1}{\overline{x}}.$$
Como $$l''(\lambda;x)=-\frac{n}{\lambda^2}<0$$ concluímos que $$\hat{\lambda}=\frac{1}{\overline{X}}.$$

\item Como $X_i \sim \text{Pois}(\lambda)$, para $0<x_i<\infty$, $i=1,\ldots,n$, a verossimilhança do parâmetro $\lambda$ será dada por 
\begin{eqnarray*}
L(\lambda;x)&=& f_{X_1}(x_1)\cdots f_{X_n}(x_n)\\
&=& \frac{e^{-\lambda}\lambda^{x_1}}{x_1!}\cdots\frac{e^{-\lambda}\lambda^{x_n}}{x_n!}\\
&=& \frac{e^{-n\lambda}\lambda^{\sum x_i}}{x_1!\cdots x_n!}.
\end{eqnarray*}
Portanto a logverossimilhança é dada por
\begin{eqnarray*}
l(\lambda;x)&=& \log e^{-n\lambda}+\log\lambda^{\sum x_i}-\log (x_1!\cdots x_n!)\\
&=& -n\lambda+\log\lambda\sum_{i=1}^{n}x_i-\log (x_1!\cdots x_n!).
\end{eqnarray*}
Derivando a logverossimilhança temos que $$l'(\lambda;x)=-n+\frac{\sum_{i=1}^{n}x_i}{\lambda}=0 \iff \lambda=\frac{\sum_{i=1}^{n}x_i}{n}=\overline{x}.$$
Como $$l''(\lambda;x)=-\frac{\sum_{i=1}^{n}x_i}{\lambda^2}<0$$ concluímos que $$\hat{\lambda}=\overline{X}.$$

\item Como $X_i\sim \mathcal{N}(\mu,\sigma^2)$, a verosimilhança dos parâmetros $\mu$ e $\sigma^2$ é
\begin{eqnarray*}
L\left(\mu,\sigma^2;x\right)&=& f_{X_1}(x_1)\cdots f_{X_n}(x_n)\\
&=& \prod_{i=1}^{n}\frac{1}{\sigma\sqrt{2\pi}}\exp\left[\frac{-(x_i-\mu)^2}{2\sigma^2}\right]\\
&=& \left(\frac{1}{2\pi}\right)^{n/2}\frac{1}{\sigma^n}\exp\left[-\sum_{i=1}^{n}\frac{(x_i-\mu)^2}{2\sigma^2}\right].\\
\end{eqnarray*}
Portanto a logverossimilhança é dada por $$l\left(\mu,\sigma^2;x\right)=-\frac{n}{2}\log(2\pi)-n\log\sigma-\frac{\sum(x_i-\mu)^2}{2\sigma^2}.$$ Derivando a logverossimilhança com relação aos parâmetros $\mu$ e $\sigma$ temos
\begin{eqnarray}
\frac{\partial l}{\partial\mu} &=& \frac{\sum(x_i-\mu)}{\sigma^2}\label{eq3}\\
\frac{\partial l}{\partial\sigma} &=& -\frac{n}{\sigma}+\frac{\sum(x_i-\mu)^2}{\sigma^3}\label{eq4}.
\end{eqnarray}
Igualando as equações (\ref{eq3}) e (\ref{eq4}) a zero e resolvendo o sistema linear para $\mu$ e $\sigma$, encontramos facilmente $$\mu=\frac{\sum_{i=1}^{n}x_i}{n} \ \ \ \text{e} \ \ \ \sigma = \sqrt{\frac{\sum_{i=1}^{n}(x_i-\mu)^2}{n}}.$$
Calculado as derivadas parciais segundas obtemos 
$$A = \frac{\partial^2 l}{\partial\mu^2} = -\frac{n}{\sigma^2}, \ \ \ B=\frac{\partial^2 l}{\partial\mu\partial\sigma}=-\frac{2\sum(x_i-\mu)}{\sigma^2} \ \ \ \text{e} \ \ \ C=\frac{\partial^2 l}{\partial\sigma^2}=\frac{n}{\sigma^2}-\frac{3\sum(x_i-\mu)^2}{\sigma^4}.$$ Como $A<0$ e $AC-B^2>0$, concluímos que $$\hat{\mu}=\frac{\sum_{i=1}^{n}X_i}{n} \ \ \ \text{e} \ \ \ \hat{\sigma} = \sqrt{\frac{\sum_{i=1}^{n}(X_i-\mu)^2}{n}}.$$
