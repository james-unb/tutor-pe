\item Primeiramente, observe que para a AAS $\widetilde{X}=(X_1,\ldots,X_{25})$ temos $X_i\sim \mathcal{N}(\mu,5^2)$ para todo $i=1,

\begin{eqnarray*}
\alpha&=& \mathcal{P}(\text{rejeitar} \ H_0 \ | \ H_0 \ \text{verdadeira})\\
&=& \mathcal{P}(\overline{X}>12 \ | \ \mu=10)\\
&=& \mathcal{P}(\overline{X}-10>12-10 \ | \ \mu=10)\\
&=& \mathcal{P}(Z>2)\\
&=& 1-\mathcal{P}(Z\leqslant 2)\\
&=& 0,02275.
\end{eqnarray*}

\noindent A probabilidade de erro tipo II é
\begin{eqnarray*}
\beta&=& \mathcal{P}(\text{não rejeitar} \ H_0 \ | \ H_0 \ \text{falsa})\\
&=& \mathcal{P}(\overline{X}\leqslant 12 \ | \ \mu=14)\\
&=& \mathcal{P}(\overline{X}-14\leqslant 12-14 \ | \ \mu=14)\\
&=& \mathcal{P}(Z\leqslant -2)\\
&=& \mathcal{P}(Z> 2)\\
&=& 0,02275.
\end{eqnarray*}

\item Do enunciado temos que $n=100$, $\overline{x}=2,1$ e $\sigma^2=0,09$.
\begin{itemize}
\item[a)] $H_0 : \mu=2 \ \ versus \ \ H_a : \mu \neq 2$.
\item[b)] Para $\alpha=0,02$ temos $z_{\alpha/2}=2,33$ e então $$\text{RC} = \{x\in\mathbb{R} : x<1,93 \ \text{ou} \ X>2,07\}.$$

\item Primeiramente, observe que para a AAS $\widetilde{X}=(X_1,\ldots,X_{50})$ temos $X_i\sim \mathcal{N}(\mu,10^2)$ para todo $i=1,

\begin{itemize}
\item[a)] Como $\overline{x}=15,2$ segue que $$z^{*}=\frac{\sqrt{n}}{\sigma}(\overline{X}-\mu_0) = \frac{\sqrt{50}}{10}(15,2-18)=-1,98$$ e portanto 
\begin{eqnarray*}
\alpha^{*} &=& \mathcal{P}(Z\leqslant z^{*}) = \mathcal{P}(Z\leqslant -1,98)\\

&=& 1-0,9761 = 0,0239.
\end{eqnarray*}
\item[b)] Análogo ao {\bf item a)}.
\item[c)] $$z^{*}=\frac{\sqrt{n}}{\sigma}|\overline{X}-\mu_0| = \frac{\sqrt{50}}{10}|15,2-18|=1,98$$ e portanto 
\begin{eqnarray*}
\alpha^{*} &=& \mathcal{P}(|Z|> z^{*}) = 2\mathcal{P}(Z>1,98)\\
&=& 2(1-\mathcal{P}(Z\leqslant 1,98)) = 2(1-0,9761)\\
&=& 0,0478.
\end{eqnarray*}
\item[d)] $$z^{*}=\frac{\sqrt{n}}{\sigma}(\overline{X}-\mu_0) = \frac{\sqrt{50}}{10}(15,2-17)=-1,27$$ e portanto 
\begin{eqnarray*}
\alpha^{*} &=& \mathcal{P}(Z\leqslant z^{*}) = \mathcal{P}(Z\leqslant -1,27)\\
&=& \mathcal{P}(Z>1,27) = 1-\mathcal{P}(Z\leqslant 1,27)\\
&=& 1-0,8979 = 0,1020.
\end{eqnarray*}
\end{itemize}

\item Do enunciado temos uma AAS $\widetilde{X}=(X_1,\ldots,X_{16})$ onde $X_i\sim \mathcal{N}(\mu \ ; \ 1,2^2)$ para todo \\ $i=1,

\item Do enunciado temos uma AAS $\widetilde{X}=(X_1,\ldots,X_{12})$ distribuída normalmente, $\overline{x}=21,7$ e $s=5,5$. \item Do enunciado temos uma AAS $\widetilde{X}=(X_1,\ldots,X_{12})$ distribuída normalmente e a partir dos dados temos $\overline{x}=2493,17$ e $s=34,45$. Estamos interessados em testar $$H_0 : \mu\geqslant 2500 \ \ versus \ \ H_a : \mu <2500.$$ Como a variância é desconhecida temos $$t^*=\frac{\sqrt{n}}{S}(\overline{X}-\mu_0)=\frac{\sqrt{12}}{34,45}(2493,17-2500) = -0,6868.$$ Desse modo

\item Estamos interessados em testar $$H_0 : p=0,9 \ \ versus \ \ H_a : p \neq 0,9.$$ À partir do dados encontramos $\widehat{p}=\frac{175}{200}$ e como $\alpha=0,1$ temos $$Z=\frac{\left |\frac{175}{200}-0,9 \right |}{\sqrt{\frac{0,9\times 0,1}{200}}} = 1,18 <  z_{0,05} = 1,64.$$ Decidimos assim pela não rejeição de $H_0$.

\vspace{0.3cm} 

\noindent Para o cálculo do p-valor basta notar que $z^* = 1,18$ e então 

\begin{eqnarray*}
\alpha^*&=& 2(1-\mathcal{P}(Z\leqslant z^*))\\
&=& 2(1-\mathcal{P}(Z\leqslant 1,18))\\
&\simeq& 2(1-0,88100)\\
&=& 0,238
\end{eqnarray*}

\noindent Como o p-valor é relativamente alto, temos fortes indícios à favor da afirmação do fabricante.

\item Estamos interessados em testar $$H_0 : p=0,6 \ \ versus \ \ H_a : p < 0,6.$$ À partir do dados encontramos $\widehat{p}=\frac{104}{200}$ e como $\alpha=0,05$ temos $$Z=\frac{\frac{104}{200}-0,6}{\sqrt{\frac{0,6\times 0,4}{200}}} = -2,31 < - z_{0,05} = -1,64.$$ Decidimos assim pela rejeição de $H_0$.

\vspace{0.3cm} 

\noindent Para o cálculo do p-valor basta notar que $z^* = -2,31$ e então 

\begin{eqnarray*}
\alpha^*&=& \mathcal{P}(Z\leqslant z^*)\\
&=& \mathcal{P}(Z\leqslant -2,31)\\
&=& 1-\mathcal{P}(Z\leqslant 2,31)\\
&\simeq& 0,01044
\end{eqnarray*}

\noindent Como o p-valor é baixo, temos fortes indícios contra a afirmação a emissora.

\item Estamos interessados em testar $$H_0 : p=0,52 \ \ versus \ \ H_a : p > 0,52.$$ À partir do dados encontramos $\widehat{p}=\frac{717}{1250}$ e $$z^*=\frac{\frac{717}{1250}-0,52}{\sqrt{\frac{0,52\times 0,48}{1250}}} = 3,79.$$ 

\noindent Segue que

\begin{eqnarray*}
\alpha^*&=& \mathcal{P}(Z\geqslant z^*)\\
&=& \mathcal{P}(Z\geqslant 3,79)\\
&=& 1-\mathcal{P}(Z\leqslant 3,79)\\
&\simeq& 0,00008
\end{eqnarray*}

\noindent Como o p-valor é muito baixo, temos fortes indícios contra a hipótese nula e concluímos que houve um aumento significativo da popularidade do governante.
