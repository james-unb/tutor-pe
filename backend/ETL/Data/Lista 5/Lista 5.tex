\item A função distribuição é $F_M(m) = [F_X(m)]^n$ e a função densidade é $f_M(m) = n[F_X(m)]^{n-1}f_X(m).$

\item Nesse caso a função densidade será
\begin{equation*}
\renewcommand{\arraystretch}{1.4}
f_M(m) = 
\left\{\begin{array}{crr}
\dfrac{nm^{n-1}}{\theta^n}, &\text{se} & m\in (0,\theta),\\
0, &\text{se} & m \notin (0,\theta).\\
\end{array} \right.
\end{equation*} 
e a função distribuição

\begin{equation*}
\renewcommand{\arraystretch}{1.4}
F_M(m) = 
\left\{\begin{array}{crr}
0, &\text{se} & m\leqslant 0,\\
\dfrac{m^n}{\theta^n}, &\text{se} & m\in (0,\theta),\\
1, &\text{se} & m \geqslant \theta.\\
\end{array} \right.
\end{equation*}
