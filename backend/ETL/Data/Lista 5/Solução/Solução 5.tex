Se $Y\sim N(23;10,02)$, aplicando o Teorema de DeMoivre-Laplace podemos aproximar a probabilidade desejada como abaixo:
\begin{eqnarray*}
\mathcal{P}(X_A+X_B\geqslant 25)&\simeq& \mathcal{P}(Y\geqslant 25)\\

\begin{eqnarray*}
\mathcal{P}(X_A-X_B\geqslant 1)&\simeq& \mathcal{P}(Y\geqslant 1)\\
&=& \mathcal{P}\left(\frac{Y+2,2}{\sqrt{10,02}}\geqslant\frac{1+2,2}{\sqrt{10,02}}\right)\\

\noindent Como $$\left(\frac{X-\mathcal{E}(X)}{\sigma(X)}-\frac{Y-\mathcal{E}(Y)}{\sigma(Y)}\right)^2\geqslant 0,$$ temos
\begin{eqnarray*}
&& \mathcal{E}\left(\frac{X-\mathcal{E}(X)}{\sigma(X)}-\frac{Y-\mathcal{E}(Y)}{\sigma(Y)}\right)^2\\
&=& \mathcal{E}\left(\frac{X-\mathcal{E}(X)}{\sigma(X)}\right)^2+\mathcal{E}\left(\frac{Y-\mathcal{E}(Y)}{\sigma(Y)}\right)^2-\frac{2}{\sigma(X)\sigma(Y)}\mathcal{E}[(X-\mathcal{E}(X))(Y-\mathcal{E}(Y))]\\
&=& \frac{\text{Var(X)}}{\sigma^2(X)}+\frac{\text{Var(Y)}}{\sigma^2(Y)}-\frac{2\text{Cov(X,Y)}}{\sigma(X)\sigma(Y)}=2-2\rho(X,Y)\\
&\geqslant& 0.
\end{eqnarray*}
Logo $\rho(X,Y)\leqslant 1.$

\noindent Substituindo ``$-$'' por ``$+$'' na expressão acima, encontramos a outra desigualdade.
\item[b)] \begin{eqnarray*}
\rho(X+a,Y+b)&=& \frac{\mathcal{E}[(X+a)(Y+b)]-\mathcal{E}(X+a)\mathcal{E}(Y+b)}{\sigma(X+a)\sigma(Y+b)}\\
&=& \frac{\mathcal{E}(XY+aY+bX+ab)-[\mathcal{E}(X)+a][\mathcal{E}(Y)+b]}{\sigma(X)\sigma(Y)}\\
&=& \frac{\mathcal{E}(XY)-\mathcal{E}(X)\mathcal{E}(Y)}{\sigma(X)\sigma(Y)}\\
&=& \rho(X,Y).
\end{eqnarray*}
\item[c)] \begin{eqnarray*}
\rho(aX,bY)&=& \frac{\mathcal{E}(aXbY)-\mathcal{E}(aX)\mathcal{E}(bY)}{\sigma(aX)\sigma(bY)}\\
&=& \frac{ab\mathcal{E}(XY)-ab\mathcal{E}(X)\mathcal{E}(Y)}{|a||b|\sigma(X)\sigma(Y)}\\
&=& \frac{ab}{|ab|}\frac{\text{Cov}(X,Y)}{\sigma(X)\sigma(Y)}\\
&=& \frac{ab}{|ab|}\rho(X,Y).
\end{eqnarray*}
\end{itemize}

&=& \mathcal{P}(X_1\leqslant m)\cdots \mathcal{P}(X_n\leqslant m)\\

\end{eqnarray*}

\end{eqnarray*}
