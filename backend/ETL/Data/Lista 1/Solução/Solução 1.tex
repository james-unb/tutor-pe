\item[a)] Se cada observação é multiplicada por 2, as estatísticas de ordem permanecem nas mesmas posições, porém multiplicadas por 2 e, portanto, a mediana é $2\tilde{x}$.

A média será dada por $$ \frac{2x_1+\cdots+2x_n}{n}=2\overline{x}.$$

O desvio padrão será $$\sqrt{\frac{(2x_1-2\overline{x})^2+\cdots+(2x_n-2\overline{x})^2}{n}}=2\sqrt{\frac{(x_1-\overline{x})^2+\cdots+(x_n-\overline{x})^2}{n}}=2\mathcal{D}(X).$$
\item[b)] Somando-se 10 unidades à mediana, a mesma apenas será transladada assim como as outras estatísticas de ordem, mantendo a posição central e, portanto, seu valor é $10+\tilde{x}$.

A média, por sua vez, será $$\frac{(10+x_1)+\cdots +(10+x_n)}{n}=\frac{10n + x_1+\cdots+x_n}{n}=10+\overline{x}.$$

Por fim, o desvio padrão será $$\sqrt{\frac{(10+x_1-(10+\overline{x}))^2+\cdots(10+x_n-(10+\overline{x}))^2}{n}}=\mathcal{D}(X).$$
\item[c)] Por argumentos análogos aos itens anteriores a mediana será $\tilde{x}-\overline{x}$.

A média dos dados será $$\frac{1}{n}\sum_{i=1}^n(x_i-\overline{x})=\frac{1}{n}\sum_{i=1}^n x_i-\overline{x}=0$$

O desvio padrão obviamente não será alterado, visto que a média acima é zero.
\item[d)] Após padronizarmos os dados teremos o conjunto $\left(\frac{x_1-\overline{x}}{\mathcal{D}(X)},\cdots,\frac{x_n-\overline{x}}{\mathcal{D}(X)}\right).$ A mediana será simplesmente dada por $\frac{\tilde{x}-\overline{x}}{\mathcal{D}(X)}$.

A média, como no caso anterior, será zero.

\newpage
O desvio padrão, por sua vez, será $$\sqrt{\frac{\left(\frac{x_1-\overline{x}}{\mathcal{D}(X)}\right)^2+\cdots+\left(\frac{x_n-\overline{x}}{\mathcal{D}(X)}\right)^2}{n}}=\frac{1}{\mathcal{D}(X)}\sqrt{\frac{(x_1-\overline{x})^2+\cdots+(x_n-\overline{x})^2}{n}}=1$$
\end{itemize}

\item Primeiramente, observe que 

\begin{eqnarray*}
\text{Var}(X) &=& \frac{\sum_{i=1}^n(x_i-\overline{x})^2}{n} = \frac{\sum_{i=1}^n \left( x_i^2-2x_i\overline{x}+\overline{x}^2\right)}{n}\\
&=& \frac{\sum_{i=1}^n x_i^2}{n} - 2\overline{x}\frac{\sum_{i=1}^n x_i}{n} + \frac{\sum_{i=1}^n \overline{x}^2}{n}\\
&=& \frac{\sum_{i=1}^n x_i^2}{n}-2\overline{x}^2+\overline{x}^2\\
&=& \frac{\sum_{i=1}^n x_i^2}{n}-\overline{x}^2\\
\end{eqnarray*}

Se separássemos os valores segundo suas frequências, teríamos
$$\text{Var}(X)=\frac{\sum_{i=1}^k n_ix_i^2}{n}-\overline{x}^2 = \sum_{i=1}^k f_ix_i^2-\overline{x}^2.$$

$$\text{Var}(x) = \frac{1}{n}\sum x_i^2-\overline{x}^2,$$ temos que

\begin{eqnarray*}
\displaystyle \frac{1}{n}\sum_i\left(\frac{x_i-\overline{x}}{\mathcal{D}(x)}\right)\left(\frac{y_i-\overline{y}}{\mathcal{D}(y)}\right) &=& \frac{1}{n}\frac{\sum (x_iy_i-\overline{x}y_i-\overline{y}x_i + \overline{x} \hspace{0.03cm} \overline{y})}{\sqrt{\left(\frac{1}{n}\sum x_i^2-\overline{x}^2\right)\left(\frac{1}{n}\sum y_i^2-\overline{y}^2\right)}}\\
&=&  \frac{1}{n}\frac{\sum x_iy_i-\overline{x}\sum y_i-\overline{y}\sum x_i + \sum\overline{x} \hspace{0.03cm} \overline{y}}{\sqrt{\left(\frac{\sum x_i^2-n\overline{x}^2}{n}\right)\left(\frac{\sum y_i^2-n\overline{y}^2}{n}\right)}}\\
&=& \frac{\sum x_iy_i-n\overline{x} \hspace{0.03cm} \overline{y}-n\overline{y} \hspace{0.03cm} \overline{x} + n\sum\overline{x} \hspace{0.03cm} \overline{y}}{\sqrt{\left(\sum x_i^2-n\overline{x}^2\right)\left(\sum y_i^2-n\overline{y}^2\right)}}\\
&=& \frac{\sum x_iy_i-n\overline{x} \hspace{0.03cm} \overline{y}}{\sqrt{\left(\sum x_i^2-n\overline{x}^2\right)\left(\sum y_i^2-n\overline{y}^2\right)}}.
\end{eqnarray*}
