\item Seja $X$ uma variável aleatória com densidade 
\begin{equation*}
f(x) = 
\left\{\begin{array}{lcc}
C(1-x^2), & \text{se} & x\in (-1,1),\\ \vspace{0.1cm}
0, & \text{se} & x \notin (-1,1).
\end{array} \right.
\end{equation*} %Ross - Capítulo 5 - Questão 5.1 - Pág. 224
\begin{itemize}
\item[a)] Qual o valor de $C$?
\item[b)] Qual a função distribuição de $X$?
\item[c)] Calcule $\mathcal{E}(X)$ e $\mathcal{E}|X|$.
\end{itemize} 

\item Seja $X$ uma variável aleatória com densidade 
\begin{equation*}
f(x) = 
\left\{\begin{array}{lcc}
Cxe^{-x/2}, & \text{se} & x>0,\\ \vspace{0.1cm}
0, & \text{se} & x \leqslant 0.
\end{array} \right.
\end{equation*} %Ross - Capítulo 5 - Questão 5.6 - Pág. 224
\begin{itemize}
\item[a)] Qual o valor de $C$?
\item[b)] Qual a função distribuição de $X$?
\item[c)] Calcule $\mathcal{E}(X)$.
\end{itemize} 

\item Seja X uma variável aleatória contínua com função densidade de probabilidades dada por:
\[
f(x) = \begin{cases}
\frac{1}{6}x+k, \quad &\text{se } \ 0\leqslant x \leqslant 3;\\
0,\quad & \text{c.c.}
\end{cases}
\]
	
\begin{itemize}
\item[a)] Determine o valor de k.
\item[b)] Calcule $\P(1<X<2)$.
\item[c)] Determine o valor de $a$ de forma que $\P(X < a) =\frac{1}{2}$. 
\end{itemize}

kx,\quad &\text{se } \ 0\leqslant x\leqslant1;\\

0,\quad &\text{c.c.}
\end{cases}
\]
\begin{itemize}
\item[a)] Qual o tempo máximo que uma peça pode ficar sem se corroer? 
\item[b)] Calcule o valor de k.
\item[c)] Calcule $\P(0,5<X<1,5)$.
\item[d)] Determine a função distribuição acumulada da variável aleatória $X$.
\end{itemize}
